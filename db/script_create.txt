-- user
CREATE TABLE "user" (
    id SERIAL PRIMARY KEY,
    nickname VARCHAR(50) NOT NULL DEFAULT 'Newborn dragon',
    about VARCHAR(200),
    login VARCHAR(50) NOT NULL UNIQUE,
    password TEXT NOT NULL, -- Хешування
    admin BOOLEAN NOT NULL DEFAULT FALSE,
    avatar TEXT DEFAULT 'https://res.cloudinary.com/dnk74mxyc/image/upload/v1784811655/default_user_zkhfkg.jpg'
);

-- book
CREATE TABLE book (
    id SERIAL PRIMARY KEY,
    original_name VARCHAR(100) NOT NULL,
    ukrainian_name VARCHAR(100) NOT NULL,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    year INT NOT NULL,
    status TEXT NOT NULL,
    volumes INT NOT NULL,
    chapters INT NOT NULL,
    extras INT NOT NULL,
    description TEXT NOT NULL,
    preview TEXT NOT NULL
);

CREATE TABLE chapters (
    id SERIAL PRIMARY KEY,
    book_id INT NOT NULL, 
    chapter_index INT NOT NULL, 
    title VARCHAR(255), 
    content_html TEXT NOT NULL, 
    CONSTRAINT fk_book FOREIGN KEY(book_id) REFERENCES book(id) ON DELETE CASCADE
);

-- INDEX FOR CHAPTERS
CREATE UNIQUE INDEX idx_book_chapter ON chapters(book_id, chapter_index);

-- WRITER
CREATE TABLE writer (
    id SERIAL PRIMARY KEY,
    full_name VARCHAR(50) NOT NULL,
    birthday DATE NOT NULL
);

-- GENRE
CREATE TABLE genre (
    id SERIAL PRIMARY KEY,
    genre TEXT NOT NULL
);

-- post
CREATE TABLE post (
    id SERIAL PRIMARY KEY,
    title VARCHAR(200) NOT NULL,
    short_description VARCHAR(250) NOT NULL,
    preview TEXT NOT NULL,
    content JSONB NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    author_id INT REFERENCES "user"(id),
    published BOOLEAN DEFAULT FALSE
);

-- TAG
CREATE TABLE tag (
    id SERIAL PRIMARY KEY,
    tag TEXT NOT NULL
);

--------- MIDDLE TABLE
-- WRITER_LIST (writer + book)
-- TABLE writer_list (book <-> writer)
CREATE TABLE writer_list (
    book_id INT NOT NULL,
    writer_id INT NOT NULL,

    PRIMARY KEY (book_id, writer_id),

    CONSTRAINT fk_writer_list_book FOREIGN KEY (book_id)
        REFERENCES book(id) ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_writer_list_writer FOREIGN KEY (writer_id)
        REFERENCES writer(id) ON UPDATE CASCADE ON DELETE CASCADE
);

-- TABLE genre_list (book <-> genre)
CREATE TABLE genre_list (
    book_id INT NOT NULL,
    genre_id INT NOT NULL,

    PRIMARY KEY (book_id, genre_id),

    CONSTRAINT fk_genre_list_book FOREIGN KEY (book_id)
        REFERENCES book(id) ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_genre_list_genre FOREIGN KEY (genre_id)
        REFERENCES genre(id) ON UPDATE CASCADE ON DELETE CASCADE
);

-- TABLE tag_list (post <-> tag)
CREATE TABLE tag_list (
    post_id INT NOT NULL,
    tag_id INT NOT NULL,

    PRIMARY KEY (post_id, tag_id),

    CONSTRAINT fk_tag_list_post FOREIGN KEY (post_id)
        REFERENCES post(id) ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_tag_list_tag FOREIGN KEY (tag_id)
        REFERENCES tag(id) ON UPDATE CASCADE ON DELETE CASCADE
);

-- TABLE watch_list (user <-> book)
CREATE TABLE watch_list (
    user_id INT NOT NULL,
    book_id INT NOT NULL,
    status TEXT NOT NULL,

	notify_new_chapters BOOLEAN DEFAULT TRUE,
    notify_completed BOOLEAN DEFAULT TRUE,

    PRIMARY KEY (user_id, book_id),

    CONSTRAINT fk_watch_list_user FOREIGN KEY (user_id)
        REFERENCES "user"(id) ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_watch_list_book FOREIGN KEY (book_id)
        REFERENCES book(id) ON UPDATE CASCADE ON DELETE CASCADE
);

-- TABLE comment
CREATE TABLE comment (
    id SERIAL PRIMARY KEY,
    parent_id INT,
    author_id INT NOT NULL,
    book_id INT,
    post_id INT,
    date TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    text TEXT NOT NULL,

    CONSTRAINT fk_comment_parent FOREIGN KEY (parent_id)
        REFERENCES comment(id) ON DELETE CASCADE,

    CONSTRAINT fk_comment_author FOREIGN KEY (author_id)
        REFERENCES "user"(id) ON DELETE CASCADE,

    CONSTRAINT fk_comment_book FOREIGN KEY (book_id)
        REFERENCES book(id) ON DELETE CASCADE,

    CONSTRAINT fk_comment_post FOREIGN KEY (post_id)
        REFERENCES post(id) ON DELETE CASCADE,

    CONSTRAINT chk_comment_target CHECK (
        (book_id IS NOT NULL AND post_id IS NULL)
        OR
        (book_id IS NULL AND post_id IS NOT NULL)
    )
);


-- RATING
CREATE TABLE rating (
    user_id INT NOT NULL,
    book_id INT NOT NULL,
    rate INT NOT NULL CHECK (rate BETWEEN 1 AND 5),

    PRIMARY KEY (user_id, book_id),

    CONSTRAINT fk_rating_user FOREIGN KEY (user_id)
        REFERENCES "user"(id)
        ON UPDATE CASCADE
        ON DELETE CASCADE,

    CONSTRAINT fk_rating_book FOREIGN KEY (book_id)
        REFERENCES book(id)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);

-- CHAT
CREATE TABLE conversation (
    id SERIAL PRIMARY KEY,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE conversation_user (
    conversation_id INT NOT NULL,
    user_id INT NOT NULL,

    PRIMARY KEY (conversation_id, user_id),

    FOREIGN KEY (conversation_id) REFERENCES conversation(id) ON DELETE CASCADE,
    FOREIGN KEY (user_id) REFERENCES "user"(id) ON DELETE CASCADE
);

CREATE TABLE message (
    id SERIAL PRIMARY KEY,
    conversation_id INT NOT NULL,
    sender_id INT NOT NULL,
    text TEXT NOT NULL,
    is_read BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (conversation_id) REFERENCES conversation(id) ON DELETE CASCADE,
    FOREIGN KEY (sender_id) REFERENCES "user"(id)
);

-- NOTIFICATION
CREATE TABLE notification (
    id SERIAL PRIMARY KEY,
    user_id INT NOT NULL,
    sender_id INT,
    book_id INT,
    comment_id INT,
    type VARCHAR(30) NOT NULL,
    is_read BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	content TEXT NOT NULL,
	
    FOREIGN KEY (user_id) REFERENCES "user"(id),
    FOREIGN KEY (sender_id) REFERENCES "user"(id),
    FOREIGN KEY (book_id) REFERENCES book(id),
    FOREIGN KEY (comment_id) REFERENCES comment(id),

    CONSTRAINT chk_notification_type CHECK (
    type IN ('NEW_CHAPTERS', 'COMPLETED', 'COMMENT_REPLY', 'NEW_MESSAGE'))
);
